# --- F-ISA/cpu/refused-cpus (R2 leaf 2) ---
# Per-CPU characterization vectors live in
# compiler/tests/test_isa_cpu_refused.py. Each refused CPU gets its
# oracle emission pinned (proving the refusal is of something real
# and distinct) plus its named refusal diagnostic.
#
# v1 (lacks native32 + fetch-atomics the route relies on):
# - oracle-base.ll emits 128B .text, no JMP32, digest pinned
#   (test_refused_cpu_oracle_base_vectors[v1])
# - oracle-corpus.ll refused by the oracle with pinned stderr
#   (test_refused_cpu_oracle_corpus_pre_v3_fails[v1])
# - request refused naming v1 (test_refused_cpu_request_names_cpu[v1])
#
# v2 (distinct pre-v3 emission, same lacks):
# - oracle-base.ll emits 120B .text (!= v1, != v3), no JMP32, pinned
#   (test_refused_cpu_oracle_base_vectors[v2])
# - oracle-corpus.ll refused by the oracle with pinned stderr
#   (test_refused_cpu_oracle_corpus_pre_v3_fails[v2])
# - request refused naming v2 (test_refused_cpu_request_names_cpu[v2])
#
# v4 (adds MEMSX the audit does not model):
# - oracle-base.ll emits 104B .text, == v3 byte-identical (the v4
#   delta is memsx-scoped), pinned
#   (test_refused_cpu_oracle_base_vectors[v4])
# - oracle-corpus.ll emits 18 words incl. pinned MEMSX word
#   0x0000000000001089 = ldxsxh r0, [r1+0]
#   (test_refused_cpu_oracle_corpus_v4_memsx)
# - request refused naming v4 (test_refused_cpu_request_names_cpu[v4])
#
# Emission guard (all three records): parsed-route emitted objects

# (v1!=v2!=v3, v3==v4): test_refused_cpu_oracle_base_pairwise.
